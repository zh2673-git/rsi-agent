        .env("REACT_INSTANCE_NAME", name)
        // v10：工具能力锚点。子实例把 WORKSPACE_ROOT 换成**用户项目目录**（这正是
        // 「新窗口」的全部意义），而看板/画像 CLI 住在代码根——不显式传，三处门控
        // 会在项目目录里找不到 scripts/board.py，于是提示词里看板段整段消失、

        .env("RSI_HOME", crate::config::workspace_dir())
