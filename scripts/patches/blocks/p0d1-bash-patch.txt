def _run_bash(args: dict) -> dict:
    command = require(args, "command")
    # cmd.exe /c 只执行首行（换行即命令终止，且不尊重引号）：多行命令的其余行会被
    # **静默丢弃且 exit_code 仍为 0**。静默截断比报错危险得多——模型会基于虚假成功
    # 继续推理。故此处 fail-loud，把问题挡在入口。
    if "\n" in command or "\r" in command:
        raise ToolError(
            "bash command 含换行符：cmd.exe /c 只执行首行，其余行会被静默丢弃（exit_code 仍为 0）。"
            "请改用单行 + `&&` / `&` 串联，或拆成多次调用；python 多语句用 `;` 分隔。",
            code="MULTILINE_UNSUPPORTED",
            field="command",
        )
    # cd 前缀容错（实测：提示词与工具描述都已明写「cwd 已是工作区根，禁止 cd 前缀」，
    # 但今日 7711 次 bash 调用里仍有 7196 次（93%）以 `cd` 开头——指令在长提示词里
    # 权重不足，靠「服从」不可靠）。故在执行层**静默剥掉**前导 cd：
    #   cd /d "D:\...\repo" && cmd   →   cmd
    # 只处理**前导** cd（不碰命令中间的 cd），剥完若为空则保持原样（不静默执行空命令）。
    # helper 定义在文件末尾——插在 `_run_bash` 之前会打断内置 P1 补丁的锚点连续性，
    # 造成两条补丁来回震荡（apply 两次即重复插入）。
    command = _strip_leading_cd(command)
