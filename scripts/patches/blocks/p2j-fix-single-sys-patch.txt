        // 需要超预算才会进入规范化裁剪路径（预算内直接幂等返回、不重排）；

            m("system", "OLD"),
            m("user", "u1"),

            m("user", "u2"),
        ];

        assert_eq!(out.iter().filter(|x| x.role == "system").count(), 1);
