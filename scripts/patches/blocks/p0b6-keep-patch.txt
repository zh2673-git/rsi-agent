pub(super) async fn maybe_compact(
        &self,
        src: &Envelope,
        session_id: &str,
        history: Vec<MemoryMsg>,
        cfg: &AgentLoopConfig,
    ) -> Vec<MemoryMsg> {
        self.maybe_compact_with_keep(src, session_id, history, cfg, cfg.compact_keep).await
    }
