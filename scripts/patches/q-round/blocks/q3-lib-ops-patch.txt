                } else {
                    let n = queue::clear(sid);
                    self.cancels.lock().unwrap().insert(sid.to_string());
                    Ok(json!({"ok": true, "session_id": sid, "cleared": n, "note": "取消信号已置位；队列已清空（前端可把待发项退回编辑器）"}))
